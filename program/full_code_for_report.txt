require_once "db.php";
if (PHP_SAPI !== 'cli') {
    DB::init_session(900);
}
ob_start(function (string $html): string {
    $token = htmlspecialchars($_SESSION['csrf_token'] ?? '', ENT_QUOTES, 'UTF-8');
    $field = '<input type="hidden" name="csrf_token" value="' . $token . '">';
    return (string)preg_replace('/(<form\b[^>]*\bmethod=["\']?post["\']?[^>]*>)/i', '$1' . $field, $html);
});
require_once "router.php";

$renderError = function (Throwable $e): void {
    if (!headers_sent()) {
        http_response_code($e instanceof PDOException ? 503 : 500);
    }
    $pageTitle    = 'Помилка системи';
    $errorTitle   = $e instanceof PDOException ? 'Помилка бази даних' : 'Помилка виконання запиту';
    $errorMessage = ($e instanceof RuntimeException)
        ? $e->getMessage()
        : 'Сервіс тимчасово недоступний або виникла внутрішня помилка. Перевірте з’єднання з БД.';
    include __DIR__ . '/views/error.php';
    exit;
};

set_exception_handler($renderError);

try {
    $path = (string)(parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/');
    $router->route($_SERVER['REQUEST_METHOD'] ?? 'GET', $path);
} catch (Throwable $e) {
    $renderError($e);
}

    document.querySelectorAll('.js-photo-input').forEach(input => {

            const errBox = input.parentElement.querySelector('.js-file-error');
            const file = input.files[0];
            if (errBox) {
                errBox.style.display = 'none';
                errBox.textContent = '';
            }
            if (!file) return;

            const maxBytes = Number(input.dataset.maxBytes || 0);
            const allowed  = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
            let msg = '';

            if (!allowed.includes(file.type)) {
                msg = 'Недопустимий формат. Оберіть JPG, PNG, WEBP або GIF.';
            } else if (maxBytes > 0 && file.size > maxBytes) {
                msg = `Файл занадто великий (${(file.size / 1024 / 1024).toFixed(1)} МБ). Максимум — ${(maxBytes / 1024 / 1024).toFixed(1)} МБ.`;
            }

            if (msg) {
                if (errBox) {
                    errBox.textContent = msg;
                    errBox.style.display = 'block';
                } else {
                    alert(msg);

    document.querySelectorAll('.pivot-modal').forEach(dialog => {

        const cellCostEl  = dialog.dataset.costTarget ? document.getElementById(dialog.dataset.costTarget) : null;
        const summaryEl   = dialog.querySelector('.pivot-modal-summary');
        const searchInput = dialog.querySelector('.pivot-search');

        const recalc = () => {
            let count = 0;
            let total = 0;
            dialog.querySelectorAll('.pivot-row').forEach(row => {
                const cb  = row.querySelector('input[type="checkbox"]');
                const qty = Number(row.querySelector('.pivot-extra')?.value || 0);
                if (cb?.checked) {
                    count++;
                    total += Number(cb.dataset.cost || 0) * qty;
                }
            });

            if (countEl) countEl.textContent = String(count);

            const budget = Number(document.querySelector('input[name="budget"]')?.value || 0);
            const diff   = budget - total;

            if (costEl) {
                costEl.innerHTML = `Сума: ${total.toLocaleString('uk-UA')} грн` +

                summaryEl.textContent = costEl
                    ? `Обрано позицій: ${count} · Сума: ${total.toLocaleString('uk-UA')} грн`
                    : `Обрано позицій: ${count}`;
            }
        };

                extra.hidden   = !cb.checked;
                extra.disabled = !cb.checked;

            extra?.addEventListener('input', recalc);

<?php
declare(strict_types=1);

class DB {
    private static ?PDO $instance  = null;
    private static ?bool $writable = null;

    public static function encrypt_session_pass(string $pass): string {
        if ($pass === '') {
            return '';
        }
        $key    = random_bytes(32);
        $iv     = random_bytes(12);
        $tag    = '';
        $cipher = openssl_encrypt($pass, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);

        setcookie('db_key', base64_encode($key), [
            'httponly' => true,
            'samesite' => 'Lax',
            'path'     => '/',
        ]);

        return base64_encode($iv . $tag . $cipher);
    }

    private static function decrypt_session_pass(): string {
        $raw    = (string)($_SESSION['db_pass'] ?? '');
        $keyB64 = (string)($_COOKIE['db_key'] ?? '');
        if ($raw === '' || $keyB64 === '') {
            return '';
        }
        $blob = base64_decode($raw, true);
        $key  = base64_decode($keyB64, true);
        if ($blob === false || $key === false || strlen($blob) < 28 || strlen($key) !== 32) {
            return '';
        }
        $iv     = substr($blob, 0, 12);
        $tag    = substr($blob, 12, 16);
        $cipher = substr($blob, 28);
        $plain  = openssl_decrypt($cipher, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);
        return $plain === false ? '' : $plain;
    }

    public static function verify_csrf(): void {
        $token = (string)($_POST['csrf_token'] ?? $_SERVER['HTTP_X_CSRF_TOKEN'] ?? '');
        if ($token === '' || !hash_equals((string)($_SESSION['csrf_token'] ?? ''), $token)) {
            http_response_code(403);
            exit('403 Forbidden: Invalid CSRF token');
        }
    }

    public static function init_session(int $ttl = 900): void {
        if (session_status() === PHP_SESSION_NONE) {
            session_set_cookie_params([
                'httponly' => true,
                'samesite' => 'Lax',
                'path'     => '/',
            ]);
            session_start();
        }
        if (empty($_SESSION['csrf_token'])) {
            $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
        }
        if (!empty($_SESSION['db_user'])) {
            $last = (int)($_SESSION['last_activity'] ?? 0);
            if ($last > 0 && (time() - $last) > $ttl) {
                $_SESSION = [];
                session_regenerate_id(true);
                $_SESSION['csrf_token']    = bin2hex(random_bytes(32));
                $_SESSION['flash_warning'] = 'Сесію завершено через 15 хвилин неактивності (виконано автоматичний вихід).';
                return;
            }
            $_SESSION['last_activity'] = time();
        }
    }

    private static function ensure_session(): void {
        if (session_status() === PHP_SESSION_NONE) {
            self::init_session();
        }
    }

    public static function connect(string $user, string $pass): PDO {
        $params = require __DIR__ . '/config.php';
        $dsn    = "{$params['rdms']}:host={$params['host']};dbname={$params['db']};charset=utf8mb4";
        return new PDO($dsn, $user, $pass, [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ]);
    }

    public static function get_connection(): PDO {
        if (self::$instance === null) {
            self::ensure_session();
            $params = require __DIR__ . '/config.php';

            if (!empty($_SESSION['db_user'])) {
                try {
                    self::$instance = self::connect((string)$_SESSION['db_user'], self::decrypt_session_pass());
                    return self::$instance;
                } catch (PDOException) {
                    unset($_SESSION['db_user'], $_SESSION['db_pass']);
                }
            }

            try {
                self::$instance = self::connect($params['guest_user'], $params['guest_pass']);
            } catch (PDOException) {
                throw new RuntimeException('Не вдалося підключитися до бази даних. Перевірте роботу сервера MariaDB та права гостьового облікового запису.');
            }
        }
        return self::$instance;
    }

    public static function is_logged_in(): bool {
        self::ensure_session();
        $params = require __DIR__ . '/config.php';
        return !empty($_SESSION['db_user']) && $_SESSION['db_user'] !== $params['guest_user'];
    }

    public static function current_user(): string {
        self::ensure_session();
        $params = require __DIR__ . '/config.php';
        return (string)($_SESSION['db_user'] ?? $params['guest_user']);
    }

    public static function can_write(): bool {
        if (self::$writable !== null) {
            return self::$writable;
        }
        try {
            $grants = self::get_connection()
                ->query('SHOW GRANTS FOR CURRENT_USER()')
                ->fetchAll(PDO::FETCH_COLUMN);

            self::$writable = false;
            foreach ($grants as $grant) {
                if (preg_match('/^GRANT\s+.*?\b(ALL PRIVILEGES|INSERT|UPDATE|DELETE)\b.*?\s+ON\s+/i', (string)$grant)) {
                    self::$writable = true;
                    break;
                }
            }
        } catch (Throwable) {
            self::$writable = false;
        }
        return self::$writable;
    }

    public static function online_stats(int $window = 180, bool $isEditing = false): array {
        self::ensure_session();
        $sid  = session_id();
        $now  = time();
        $path = sys_get_temp_dir() . '/expeditions_db_presence.json';

        $fp = @fopen($path, 'c+');
        if (!$fp) {
            $rw = self::can_write();
            return ['online' => 1, 'rw' => $rw ? 1 : 0, 'editing' => ($rw && $isEditing) ? 1 : 0];
        }

        flock($fp, LOCK_EX);
        $raw  = stream_get_contents($fp);
        $data = $raw ? (json_decode($raw, true) ?: []) : [];

        foreach ($data as $k => $entry) {
            if (($now - (int)($entry['ts'] ?? 0)) > $window) {
                unset($data[$k]);
            }
        }

        if ($sid !== '') {
            $canWrite   = self::can_write();
            $data[$sid] = [
                'ts'      => $now,
                'rw'      => $canWrite,
                'editing' => $canWrite && $isEditing,
            ];
        }

        ftruncate($fp, 0);
        rewind($fp);
        fwrite($fp, json_encode($data));
        fflush($fp);
        flock($fp, LOCK_UN);
        fclose($fp);

        $online  = count($data);
        $rw      = 0;
        $editing = 0;
        foreach ($data as $entry) {
            if (!empty($entry['rw'])) $rw++;
            if (!empty($entry['editing'])) $editing++;
        }

        return ['online' => $online, 'rw' => $rw, 'editing' => $editing];
    }

    public static function remove_presence(): void {
        $sid  = session_id();
        $path = sys_get_temp_dir() . '/expeditions_db_presence.json';
        if ($sid === '' || !file_exists($path)) {
            return;
        }

        $fp = @fopen($path, 'c+');
        if (!$fp) {
            return;
        }

        flock($fp, LOCK_EX);
        $raw  = stream_get_contents($fp);
        $data = $raw ? (json_decode($raw, true) ?: []) : [];
        unset($data[$sid]);
        ftruncate($fp, 0);
        rewind($fp);
        fwrite($fp, json_encode($data));
        fflush($fp);
        flock($fp, LOCK_UN);
        fclose($fp);
    }
}

     * @param array<string, array<string, callable>> $routes
     */
    public function __construct(
        private array $routes = [],
    ) {}

    /**
     * @param callable(): mixed $handler
     */
    public function add(string $method, string $path, callable $handler): void
    {
        $this->routes[$method][$path] = $handler;
    }

    public function route(string $method, string $path): void
    {
        $handler = $this->routes[$method][$path] ?? null;

        if (!$handler) {
            $pageTitle = "Page not found";
            http_response_code(404);
            include 'views/404.php';
            return;
        }

        $handler();
    }
}

function parse_size_bytes(string $val): int {
    $val  = trim($val);
    $unit = strtolower($val[strlen($val) - 1] ?? '');

function col_bounds(string $sqlType): ?array {
    $unsigned = str_contains(strtolower($sqlType), 'unsigned');

    if (preg_match('/^decimal\((\d+),(\d+)\)/i', $sqlType, $m)) {
        $precision = (int)$m[1];
        $scale     = (int)$m[2];
        $intDigits = max(1, $precision - $scale);
        $maxStr    = str_repeat('9', $intDigits) . ($scale > 0 ? '.' . str_repeat('9', $scale) : '');
        $step      = $scale > 0 ? '0.' . str_repeat('0', $scale - 1) . '1' : '1';
        return [
            'min'  => $unsigned ? '0' : "-$maxStr",
            'max'  => $maxStr,
            'step' => $step,
        ];
    }

    $intLimits = [
        'tinyint'   => [$unsigned ? '0' : '-128', $unsigned ? '255' : '127'],
        'smallint'  => [$unsigned ? '0' : '-32768', $unsigned ? '65535' : '32767'],
        'mediumint' => [$unsigned ? '0' : '-8388608', $unsigned ? '16777215' : '8388607'],
        'int'       => [$unsigned ? '0' : '-2147483648', $unsigned ? '4294967295' : '2147483647'],
        'bigint'    => [$unsigned ? '0' : '-9223372036854775808', $unsigned ? '18446744073709551615' : '9223372036854775807'],
    ];

    foreach ($intLimits as $type => [$min, $max]) {
        if (preg_match("/^$type\b/i", $sqlType)) {
            return ['min' => $min, 'max' => $max, 'step' => '1'];
        }
    }

    return null;
}

function pk_where(string|array $pk, string $rawId): array {
    $cols   = (array)$pk;
    $vals   = explode(':', $rawId, count($cols));
    $where  = [];
    $params = [];

    foreach ($cols as $i => $col) {
        $where[]           = ident($col) . " = :__pk_$i";
        $params["__pk_$i"] = $vals[$i] ?? '';
    }

    return [implode(' AND ', $where), $params];
}

function redirect(string $uri, ?string $error = null): void {
    $sep = str_contains($uri, '?') ? '&' : '?';
    header('Location: ' . $uri . ($error !== null ? $sep . 'error=' . urlencode($error) : ''));
    exit;
}

function handle_uploads(string $table): array {
    $uploaded     = [];
    $allowedMimes = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];

    foreach ($_FILES as $field => $file) {
        if (($file['error'] ?? UPLOAD_ERR_NO_FILE) !== UPLOAD_ERR_OK) continue;
        $maxBytes = max_upload_bytes();
        if (($file['size'] ?? 0) > $maxBytes) {
            $maxMb = round($maxBytes / 1024 / 1024, 1);
            throw new RuntimeException("Розмір фото перевищує дозволений ліміт ({$maxMb} МБ).");
        }

        $mime = (new finfo(FILEINFO_MIME_TYPE))->file($file['tmp_name']);
        $ext  = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        if (!in_array($mime, $allowedMimes, true) || !in_array($ext, ['jpg', 'jpeg', 'png', 'webp', 'gif'], true)) {
            throw new RuntimeException('Дозволені лише зображення (JPG, PNG, WEBP, GIF).');
        }

        $relPath = "uploads/$table/" . uniqid('', true) . ".$ext";
        $absPath = __DIR__ . "/$relPath";
        if (!is_dir(dirname($absPath))) {
            mkdir(dirname($absPath), 0775, true);
        }

        if (move_uploaded_file($file['tmp_name'], $absPath)) {
            $uploaded[$field] = $relPath;
        }
    }
    return $uploaded;
}

<?php
declare(strict_types=1);

require_once "helpers.php";

function get_foreign(PDO $dbc, string $refTable, string $labelCol): array {
    $t    = ident($refTable);
    $c    = ident($labelCol);
    $rows = $dbc->query("SELECT id, $c FROM $t")->fetchAll(PDO::FETCH_KEY_PAIR);

    $res = [];
    foreach ($rows as $id => $label) {
        $res[$id] = [
            'label' => $label,
            'url'   => "/$refTable?id=$id",
        ];
    }
    return $res;
}

function select(string $table, string|array $pk, array $columns, array $fixedWhere = []): array {
    $dbc    = DB::get_connection();
    $pkCols = (array)$pk;

    $selectParts = [];
    foreach ($pkCols as $i => $col) {
        $selectParts[] = ident($col) . " AS `__pk_$i`";
    }

    $fks        = [];
    $whereParts = [];
    $params     = [];
    $sortExprs  = [];
    $sortMap    = [];

    foreach ($columns as $field => $meta) {
        if ($meta instanceof Pivot) continue;
        $header = is_string($meta) ? $meta : $meta->label;

        if ($meta instanceof Foreign) {
            $selectParts[]      = ident($field);
            $fks[$field]        = get_foreign($dbc, $meta->table, $meta->column);
            $sortExprs[$field]  = ident($field);
            $sortMap[$header]   = $field;
        } elseif ($meta instanceof Subquery) {
            $selectParts[] = "({$meta->sql}) AS " . ident($field);
            if ($meta->table === null) {
                $sortExprs[$field] = $meta->sortSql !== null ? "({$meta->sortSql})" : ident($field);
                $sortMap[$header]  = $field;
            } else {
                $sortMap[$header] = null;
            }
        } else {
            $selectParts[] = ident($field);
            if (!str_contains($field, '_path')) {
                $sortExprs[$field] = ident($field);
                $sortMap[$header]  = $field;
            } else {
                $sortMap[$header] = null;
            }
        }
    }

    if (!empty($fixedWhere)) {
        foreach ($fixedWhere as $col => $val) {
            $whereParts[]         = ident($col) . " = :fixed_$col";
            $params["fixed_$col"] = $val;
        }
    } else {
        if (is_string($pk) && isset($_GET['id']) && $_GET['id'] !== '') {
            $whereParts[] = ident($pk) . ' = :id';
            $params['id'] = (int)$_GET['id'];
        }

        $q = trim((string)($_GET['q'] ?? ''));
        if ($q !== '') {
            $likeParts = [];
            $idx       = 0;
            foreach ($columns as $field => $meta) {
                if (is_string($meta) && !str_contains($field, '_path')) {
                    $paramKey          = "q_$idx";
                    $likeParts[]       = ident($field) . " LIKE :$paramKey";
                    $params[$paramKey] = "%$q%";
                    $idx++;
                }
            }
            if ($likeParts) {
                $whereParts[] = '(' . implode(' OR ', $likeParts) . ')';
            }
        }

        foreach ($columns as $field => $meta) {
            if ($meta instanceof Foreign && isset($_GET[$field]) && $_GET[$field] !== '') {
                $whereParts[]       = ident($field) . " = :f_$field";
                $params["f_$field"] = (int)$_GET[$field];
            }
        }
    }

    $where     = $whereParts ? 'WHERE ' . implode(' AND ', $whereParts) : '';
    $sortField = (string)($_GET['sort'] ?? '');
    $sortDir   = strtoupper((string)($_GET['dir'] ?? 'ASC')) === 'DESC' ? 'DESC' : 'ASC';

    if ($sortField !== '' && isset($sortExprs[$sortField])) {
        $orderBy = 'ORDER BY ' . $sortExprs[$sortField] . " $sortDir";
    } else {
        $orderCols = array_map(fn($c) => ident($c) . " $sortDir", $pkCols);
        $orderBy   = $orderCols ? 'ORDER BY ' . implode(', ', $orderCols) : '';
    }

    $attributes = implode(', ', $selectParts);
    $tableIdent = ident($table);

    $query = $dbc->prepare("SELECT $attributes FROM $tableIdent $where $orderBy");
    $query->execute($params);
    $items = $query->fetchAll(PDO::FETCH_ASSOC);
    $query->closeCursor();

    $result = [];
    foreach ($columns as $field => $meta) {
        if ($meta instanceof Pivot) continue;
        $header = is_string($meta) ? $meta : $meta->label;
        $values = array_column($items, $field);

        if (isset($fks[$field])) {
            $values = array_map(fn($id) => $fks[$field][$id] ?? 'Не визначено', $values);
        } elseif ($meta instanceof Subquery && $meta->table !== null) {
            $values = array_map(function ($json) use ($meta) {
                $rows = $json ? json_decode((string)$json, true) : [];
                return array_map(fn($r) => [
                    'label' => $r['label'],
                    'url'   => "/{$meta->table}?id={$r['id']}",
                ], $rows ?: []);
            }, $values);
        }

        $result[$header] = $values;
    }

    if ($pkCols) {
        $result['__id'] = array_map(
            fn($row) => implode(':', array_map(fn($i) => (string)$row["__pk_$i"], array_keys($pkCols))),
            $items
        );
    }

    $result['__sort'] = $sortMap;
    return $result;
}

function get_scheme(string $table): array {
    $dbc = DB::get_connection();
    return $dbc->query('DESCRIBE ' . ident($table))->fetchAll(PDO::FETCH_ASSOC);
}

function find_one(string $table, string|array $pk, string $rawId): ?array {
    $dbc = DB::get_connection();
    [$whereSql, $params] = pk_where($pk, $rawId);
    $stmt = $dbc->prepare('SELECT * FROM ' . ident($table) . " WHERE $whereSql");
    $stmt->execute($params);
    return $stmt->fetch(PDO::FETCH_ASSOC) ?: null;
}

function save_row(string $table, string|array $pk, array $columns, array $data, ?string $rawId = null): void {
    $dbc    = DB::get_connection();
    $scheme = get_scheme($table);
    $data   = array_merge($data, handle_uploads($table));

    $values = [];
    foreach ($scheme as $col) {
        $f = $col['Field'];
        if (str_contains($col['Extra'], 'auto_increment') || !array_key_exists($f, $data)) continue;

        $raw = is_string($data[$f]) ? trim($data[$f]) : $data[$f];

        if ($raw === '' || $raw === null) {
            if ($col['Null'] === 'NO' && $col['Default'] === null) {
                throw new RuntimeException("Поле «{$f}» є обов'язковим для заповнення.");
            }
            $values[$f] = null;
            continue;
        }

        if (str_starts_with($col['Type'], 'enum(') && !in_array($raw, parse_enum($col['Type']), true)) {
            throw new RuntimeException("Недопустиме значення для поля «{$f}».");
        }

        $bounds = col_bounds($col['Type']);
        if ($bounds !== null) {
            if (!is_numeric($raw) || (float)$raw < (float)$bounds['min'] || (float)$raw > (float)$bounds['max']) {
                throw new RuntimeException("Значення поля «{$f}» виходить за межі [{$bounds['min']} .. {$bounds['max']}].");
            }
        }

        if (preg_match('/^varchar\((\d+)\)/i', $col['Type'], $m) && mb_strlen((string)$raw) > (int)$m[1]) {
            throw new RuntimeException("Поле «{$f}» перевищує максимум у {$m[1]} символів.");
        }
        if ($f === 'phone' && $raw !== null) {
            $normalized = preg_replace('/[\s\-\(\)]+/', '', (string)$raw);
            if (!preg_match('/^(\+380\d{9}|\+(?!380)[1-9]\d{9,14})$/', $normalized)) {
                throw new RuntimeException('Некоректний номер телефону. Використовуйте формат +380XXXXXXXXX (9 цифр після +380).');
            }
            $raw = $normalized;
        }

        $values[$f] = $raw;
    }

    if (!empty($values['start_date']) && !empty($values['end_date']) && $values['end_date'] < $values['start_date']) {
        throw new RuntimeException('Дата завершення експедиції не може бути раніше дати початку.');
    }

    $dbc->beginTransaction();
    try {
        if ($rawId === null || $rawId === '') {
            $cols         = implode(', ', array_map('ident', array_keys($values)));
            $placeholders = implode(', ', array_map(fn($k) => ":$k", array_keys($values)));
            $sql          = 'INSERT INTO ' . ident($table) . " ($cols) VALUES ($placeholders)";
            $dbc->prepare($sql)->execute($values);
            $parentId = (string)$dbc->lastInsertId();
        } else {
            [$whereSql, $pkParams] = pk_where($pk, $rawId);
            $sets = implode(', ', array_map(fn($k) => ident($k) . " = :$k", array_keys($values)));
            $sql  = 'UPDATE ' . ident($table) . " SET $sets WHERE $whereSql";
            $dbc->prepare($sql)->execute(array_merge($values, $pkParams));
            $parentId = $rawId;
        }

        foreach ($columns as $field => $meta) {
            if (!($meta instanceof Pivot)) continue;

            $pivotScheme = array_column(get_scheme($meta->pivotTable), null, 'Field');
            $extraCol    = $pivotScheme[$meta->extraField] ?? null;

            $del = $dbc->prepare('DELETE FROM ' . ident($meta->pivotTable) . ' WHERE ' . ident($meta->localKey) . ' = :pid');
            $del->execute(['pid' => $parentId]);

            $submitted = is_array($data[$field] ?? null) ? $data[$field] : [];
            $ins = $dbc->prepare(
                'INSERT INTO ' . ident($meta->pivotTable) .
                ' (' . ident($meta->localKey) . ', ' . ident($meta->foreignKey) . ', ' . ident($meta->extraField) . ')' .
                ' VALUES (:pid, :fid, :extra)'
            );

            foreach ($submitted as $targetId => $row) {
                if (empty($row['checked'])) continue;

                $extraVal = trim((string)($row['extra'] ?? ''));
                if ($extraVal === '') {
                    throw new RuntimeException("Вкажіть «{$meta->extraLabel}» для всіх обраних позицій.");
                }

                if ($extraCol) {
                    if (str_starts_with($extraCol['Type'], 'enum(') && !in_array($extraVal, parse_enum($extraCol['Type']), true)) {
                        throw new RuntimeException("Недопустиме значення «{$meta->extraLabel}».");
                    }
                    $b = col_bounds($extraCol['Type']);
                    if ($b !== null && (!is_numeric($extraVal) || (float)$extraVal < max(1, (float)$b['min']) || (float)$extraVal > (float)$b['max'])) {
                        throw new RuntimeException("Кількість у «{$meta->label}» має бути від 1 до {$b['max']}.");
                    }
                }

                $ins->execute([
                    'pid'   => (int)$parentId,
                    'fid'   => (int)$targetId,
                    'extra' => $extraVal,
                ]);
            }
        }

        $dbc->commit();
    } catch (Throwable $e) {
        if ($dbc->inTransaction()) $dbc->rollBack();
        if ($e instanceof PDOException && $e->getCode() === '23000') {
            throw new RuntimeException('Порушено цілісність даних або дублікат запису.');
        }
        throw $e;
    }
}

function delete_row(string $table, string|array $pk, string $rawId): void {

    [$whereSql, $params] = pk_where($pk, $rawId);
    $stmt = $dbc->prepare('DELETE FROM ' . ident($table) . " WHERE $whereSql");
